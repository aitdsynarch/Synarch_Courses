export const slugify = (text) =>
  text
    .toLowerCase()
    .replace(/&/g, "and")
    .replace(/[^a-z0-9]+/g, "-")
    .replace(/^-|-$/g, "");

export default function HtmlNav({ onTopicClick, activeTopic }) {
  const [mobileOpen, setMobileOpen] = useState(false);
  const [activeDropdown, setActiveDropdown] = useState(null);

  const closeMenus = () => {
    setMobileOpen(false);
    setActiveDropdown(null);
  };

  const selectTopic = (topic) => {
    onTopicClick?.(topic);
    closeMenus();
  };

  const renderTopics = (section) => (
    <div className="max-h-[65vh] overflow-y-auto p-2">
      {section.topics.map((topic) => (
        <button
          key={topic}
          type="button"
          onClick={() => selectTopic(topic)}
          className={`block w-full rounded-lg px-3 py-2.5 text-left text-sm transition ${
            activeTopic === slugify(topic)
              ? "bg-orange-50 font-semibold text-orange-600"
              : "text-slate-600 hover:bg-slate-50 hover:text-orange-600"
          }`}
        >
          {topic}
        </button>
      ))}
    </div>
  );

  return (
    <nav
      className="sticky top-0 z-50 border-b border-slate-200 bg-white shadow-sm"
      onKeyDown={(event) => {
        if (event.key === "Escape") closeMenus();
      }}
    >
      <div
        className="relative mx-auto max-w-[1600px] px-4 sm:px-6"
        onMouseLeave={() => setActiveDropdown(null)}
      >
        {/* Header */}
        <div className="flex min-h-[68px] items-center justify-between gap-4">
          <button
            type="button"
            onClick={() => selectTopic("What is HTML?")}
            className="flex shrink-0 items-center gap-2 text-left"
          >
            <span className="flex h-10 w-10 items-center justify-center rounded-xl bg-orange-500 text-white">
              <Code2 size={23} />
            </span>

            <span>
              <span className="block text-base font-bold text-slate-900">
                HTML5
              </span>
              <span className="block text-xs text-slate-500">
                Zero to Advanced
              </span>
            </span>
          </button>

          {/* Desktop horizontal navigation */}
          <div className="hidden min-w-0 flex-1 lg:block">
            <div
              className="html-horizontal-scroll flex items-center gap-1 overflow-x-auto whitespace-nowrap px-1"
              onWheel={(event) => {
                if (event.deltaY !== 0) {
                  event.currentTarget.scrollLeft += event.deltaY;
                }
              }}
            >
              {htmlTopics.map((section, index) => (
                <button
                  key={section.title}
                  type="button"
                  aria-expanded={activeDropdown === index}
                  onMouseEnter={() => setActiveDropdown(index)}
                  onFocus={() => setActiveDropdown(index)}
                  onClick={() =>
                    setActiveDropdown(
                      activeDropdown === index ? null : index
                    )
                  }
                  className={`flex shrink-0 items-center gap-1 rounded-lg px-3 py-3 text-sm font-medium transition ${
                    activeDropdown === index
                      ? "bg-orange-50 text-orange-600"
                      : "text-slate-600 hover:bg-slate-50 hover:text-orange-600"
                  }`}
                >
                  {section.title}
                  <ChevronDown
                    size={15}
                    className={`shrink-0 transition-transform ${
                      activeDropdown === index ? "rotate-180" : ""
                    }`}
                  />
                </button>
              ))}
            </div>
          </div>

          {/* Mobile menu toggle */}
          <button
            type="button"
            aria-label={mobileOpen ? "Close menu" : "Open menu"}
            aria-expanded={mobileOpen}
            onClick={() => {
              setMobileOpen((open) => !open);
              setActiveDropdown(null);
            }}
            className="flex h-10 w-10 shrink-0 items-center justify-center rounded-lg border border-slate-200 text-slate-700 hover:bg-slate-50 lg:hidden"
          >
            {mobileOpen ? <X size={22} /> : <Menu size={22} />}
          </button>
        </div>

        {/* Desktop dropdown - outside the scroll container */}
        {activeDropdown !== null && (
          <div
            className="absolute left-0 right-0 top-full z-[100] hidden border-t border-slate-200 bg-white shadow-xl lg:block"
            onMouseEnter={() => setActiveDropdown(activeDropdown)}
          >
            <div className="mx-auto max-w-[1600px] px-6 py-5">
              <div className="mb-3 flex items-center gap-2">
                <BookOpen size={18} className="text-orange-500" />
                <h3 className="text-sm font-bold text-slate-900">
                  {htmlTopics[activeDropdown].title}
                </h3>
                <span className="text-xs text-slate-400">
                  {htmlTopics[activeDropdown].topics.length} lessons
                </span>
              </div>

              <div className="grid max-h-[55vh] grid-cols-2 gap-2 overflow-y-auto md:grid-cols-3 xl:grid-cols-4">
                {htmlTopics[activeDropdown].topics.map((topic) => (
                  <button
                    key={topic}
                    type="button"
                    onClick={() => selectTopic(topic)}
                    className={`rounded-lg px-3 py-2.5 text-left text-sm transition ${
                      activeTopic === slugify(topic)
                        ? "bg-orange-50 font-semibold text-orange-600"
                        : "text-slate-600 hover:bg-slate-50 hover:text-orange-600"
                    }`}
                  >
                    {topic}
                  </button>
                ))}
              </div>
            </div>
          </div>
        )}

        {/* Mobile accordion */}
        {mobileOpen && (
          <div className="border-t border-slate-100 py-3 lg:hidden">
            <div className="max-h-[75vh] space-y-2 overflow-y-auto pb-2">
              {htmlTopics.map((section, index) => (
                <div
                  key={section.title}
                  className="overflow-hidden rounded-xl border border-slate-200"
                >
                  <button
                    type="button"
                    aria-expanded={activeDropdown === index}
                    onClick={() =>
                      setActiveDropdown(
                        activeDropdown === index ? null : index
                      )
                    }
                    className={`flex w-full items-center justify-between gap-3 px-4 py-3 text-left text-sm font-semibold transition ${
                      activeDropdown === index
                        ? "bg-orange-50 text-orange-700"
                        : "bg-white text-slate-800 hover:bg-slate-50"
                    }`}
                  >
                    <span>{section.title}</span>
                    <ChevronDown
                      size={17}
                      className={`shrink-0 transition-transform ${
                        activeDropdown === index ? "rotate-180" : ""
                      }`}
                    />
                  </button>

                  {activeDropdown === index && (
                    <div className="border-t border-slate-100 bg-slate-50">
                      {renderTopics(section)}
                    </div>
                  )}
                </div>
              ))}
            </div>
          </div>
        )}
      </div>

      <style>{`
        .html-horizontal-scroll {
          scrollbar-width: thin;
          scrollbar-color: #cbd5e1 transparent;
          overscroll-behavior-x: contain;
          -webkit-overflow-scrolling: touch;
        }

        .html-horizontal-scroll::-webkit-scrollbar {
          height: 5px;
        }

        .html-horizontal-scroll::-webkit-scrollbar-track {
          background: transparent;
        }

        .html-horizontal-scroll::-webkit-scrollbar-thumb {
          background: #cbd5e1;
          border-radius: 999px;
        }

        .html-horizontal-scroll::-webkit-scrollbar-thumb:hover {
          background: #94a3b8;
        }
      `}</style>
    </nav>
  );
}
